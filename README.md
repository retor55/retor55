## Моя любимая сортировка :
'''c++
void QuickSort(int a[], int left, int right) {
  if (left >= right) {
    return;
  }
  int pivot = a[left + rand() % (right - left + 1)];
  int i = left, j = right;
  while (i <= j) {
    while (a[i] < pivot) {
      ++i;
    }
    while (a[j] > pivot) {
      --j;
    }
    if (i <= j) {
      int key = a[i];
      a[i] = a[j];
      a[j] = key;
      ++i;
      --j;
    }
  }
  QuickSort(a, left, j);
  QuickSort(a, i, right);
}
'''
